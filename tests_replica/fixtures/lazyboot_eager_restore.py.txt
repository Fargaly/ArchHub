@with_restore_relation_projection_scope
def restore_universal_application(
    map_path: str | Path,
    store: CellStore,
    *,
    key_provider: SigningKeyProvider,
    court_workspace_root: str | Path | None = None,
    runtime_compliance_runner: Callable[
        [CourtInvocation], CourtResult
    ] | None = None,
) -> tuple[CellStore, UniversalApplicationRegistry]:
    """Recover the live application registry from one durable Cell graph."""
    snapshot = store.snapshot()
    application_root = "app:archhub"
    if application_root not in snapshot.cells:
        raise InvalidCell("durable universal application root is missing")
    if _project_legacy_baboom_execution_catalog_entry(snapshot) is not None:
        raise InvalidCell(
            "legacy duplicate BABOOM Agent Body requires a controlled graph "
            "migration before normal restore"
        )
    presentation_vocabulary = (
        Cell(
            "app:role:presentation-binding",
            NULL_CELL_ID,
            NULL_CELL_ID,
            b"presentation-binding",
        ),
        Cell(
            "app:role:contract",
            NULL_CELL_ID,
            NULL_CELL_ID,
            b"contract",
        ),
        Cell(
            "app:role:previous",
            NULL_CELL_ID,
            NULL_CELL_ID,
            b"previous",
        ),
        Cell(
            "app:role:migration",
            NULL_CELL_ID,
            NULL_CELL_ID,
            b"migration",
        ),
        Cell(
            _PRESENTATION_FALLBACK_MODE_ROOT,
            NULL_CELL_ID,
            NULL_CELL_ID,
            b"fallback when the versioned value is empty",
        ),
        Cell(
            _PRESENTATION_COLOR_CONTRACT_ROOT,
            NULL_CELL_ID,
            NULL_CELL_ID,
            b"empty fallback or six-digit sRGB color",
        ),
    )
    missing_presentation_vocabulary = tuple(
        cell for cell in presentation_vocabulary
        if cell.id not in snapshot.cells
    )
    if missing_presentation_vocabulary:
        store.commit(
            snapshot.revision, create=missing_presentation_vocabulary
        )
        snapshot = store.snapshot()
    map_registry = project_grand_map_cells(snapshot, map_path)
    roles = dict(map_registry.roles)
    roles.update({
        name: "app:role:%s" % name
        for name in (
            "focus", "selected", "available", "authority", "seed",
            "property", "relation", "lens", "catalog", "read-only",
            "visible", "binding-mode", "presentation-binding", "contract",
            "previous", "migration",
        )
    })
    if any(root_id not in snapshot.cells for root_id in roles.values()):
        raise InvalidCell("persisted application role vocabulary is incomplete")

    relation_composer_protocol = _ensure_relation_composer_protocol_graph(
        store, member_role_id=roles["member"]
    )
    snapshot = store.snapshot()

    assembly = project_assembly_protocol(snapshot, "app:assembly-protocol")
    _ensure_standard_library_state_machine_protocol_roles(store)
    snapshot = store.snapshot()
    legacy_standard_library = _project_migratable_standard_library(
        snapshot, assembly
    )
    adapter = project_adapter_protocol(snapshot, "app:adapter-protocol")
    composer = project_composer_protocol(snapshot, "app:composer-protocol")
    authorization = _restore_application_authorization(store, key_provider)
    ensure_tenant_release_descriptor_protocol(
        store, prefix="app:tenant-configuration-protocol"
    )
    snapshot = store.snapshot()
    tenant_configuration_protocol = project_tenant_configuration_protocol(
        snapshot, prefix="app:tenant-configuration-protocol"
    )
    attestation = project_attestation_protocol(
        snapshot, "app:attestation-protocol"
    )
    tenant_authority_court_root = "app:court:tenant-authority"
    verify_court_definition(
        snapshot, attestation, tenant_authority_court_root
    )
    attestation_broker = CourtAttestationBroker(
        key_provider=key_provider,
        key_id="archhub.local.court-attestation",
    )
    attestation_broker.admit_court(
        snapshot,
        attestation,
        tenant_authority_court_root,
        _tenant_authority_court_runner(frozenset((
            legacy_standard_library.lifecycle_protocol.states["shared"],
            legacy_standard_library.lifecycle_protocol.states["published"],
        ))),
    )
    authority_migration = _migrate_universal_authorities_current(
        store,
        assembly,
        legacy_standard_library,
        adapter,
        composer,
        tenant_configuration_protocol,
        authorization,
        attestation,
        attestation_broker,
        tenant_authority_court_root,
        roles,
        models_domain_root=map_registry.domains["models"],
    )
    standard_library = authority_migration.standard_library
    cognition_protocol = authority_migration.cognition_protocol
    cognition_definitions = authority_migration.cognition_definitions
    adapter_catalog_root = authority_migration.adapter_catalog_root
    device_custody_adapter_root = (
        authority_migration.device_custody_adapter_root
    )
    composer_authority = authority_migration.composer_authority
    snapshot = store.snapshot()
    _ensure_node_library_sections(
        store,
        assembly,
        roles,
        standard_library,
        cognition_definitions,
        library_root="app:node-library",
    )
    snapshot = store.snapshot()
    tenant_title_relation = "app:property:%s:field:title" % _part(
        authorization.tenant_root
    )
    if tenant_title_relation not in snapshot.cells:
        tenant_title_value = "app:value:%s:field:title" % _part(
            authorization.tenant_root
        )
        title_relation = compose_relation_cells(
            (
                (roles["owner"], authorization.tenant_root),
                (roles["value"], tenant_title_value),
                (roles["label"], "app:label:title"),
            ),
            relation_id=tenant_title_relation,
        )
        canvas_patch = prepare_append_relation_members(
            snapshot,
            "app:canvas",
            ((roles["property"], tenant_title_relation),),
            budget=100_000,
        )
        lens_patch = prepare_append_relation_members(
            snapshot,
            "app:properties-lens",
            ((roles["scope"], tenant_title_relation),),
            budget=100_000,
        )
        store.commit(
            snapshot.revision,
            create=(
                Cell(
                    tenant_title_value,
                    NULL_CELL_ID,
                    NULL_CELL_ID,
                    b"ArchHub founder workspace",
                ),
                *title_relation.cells,
                *canvas_patch.create,
                *lens_patch.create,
            ),
            replace=(*canvas_patch.replace, *lens_patch.replace),
        )
        snapshot = store.snapshot()
    cloud_route_protocol = project_cloud_route_protocol(
        snapshot, prefix="app:cloud-route-protocol"
    )
    existing_route_keys = {
        "%s %s" % (route.method, route.path_template)
        for route in list_cloud_routes(snapshot, cloud_route_protocol)
    }
    missing_route_specs = tuple(
        spec for spec in _APPLICATION_HTTP_ROUTE_SPECS
        if "%s %s" % spec[:2] not in existing_route_keys
    )
    if missing_route_specs:
        restored_route_roots = []
        for method, path_template, action_name in missing_route_specs:
            restored_route_roots.append(build_cloud_route(
                store,
                cloud_route_protocol,
                route_id="app:http-route:%s:%s" % (
                    method.lower(), _part(path_template)
                ),
                method=method,
                path_template=path_template,
                action_root=authorization.protocol.actions[action_name],
                object_root=authorization.route_scope_root,
                interface_root="app:ui-protocol:root",
                purpose_root=authorization.purpose_root,
                audience_root=authorization.audience_root,
                classification_root=authorization.classification_root,
                lifecycle_state_root=(
                    standard_library.lifecycle_protocol.states["published"]
                ),
                resource_lineage_roots=(
                    authorization.scope_root,
                    authorization.route_scope_root,
                ),
            ))
        route_snapshot = store.snapshot()
        application_patch = prepare_append_relation_members(
            route_snapshot,
            application_root,
            (
                (roles["member"], route_root)
                for route_root in restored_route_roots
            ),
            budget=100_000,
        )
        store.commit(
            route_snapshot.revision,
            create=application_patch.create,
            replace=application_patch.replace,
        )
        snapshot = store.snapshot()
        cloud_route_protocol = project_cloud_route_protocol(
            snapshot, prefix="app:cloud-route-protocol"
        )
    projected_routes = {
        "%s %s" % (route.method, route.path_template): route.root_id
        for route in list_cloud_routes(snapshot, cloud_route_protocol)
    }
    expected_route_keys = {
        "%s %s" % (method, path)
        for method, path, _ in _APPLICATION_HTTP_ROUTE_SPECS
    }
    expected_active_route_keys = (
        expected_route_keys - _RETIRED_APPLICATION_HTTP_ROUTE_KEYS
    )
    website_route_keys = {
        "GET %s" % path for path in PUBLIC_WEBSITE_ROUTES
    }
    active_route_keys = (
        set(projected_routes) - _RETIRED_APPLICATION_HTTP_ROUTE_KEYS
    )
    core_website_route_keys = {
        "GET %s" % path for path in CORE_WEBSITE_ROUTES
    }
    if active_route_keys not in (
        expected_active_route_keys,
        expected_active_route_keys | website_route_keys,
        expected_active_route_keys | core_website_route_keys,
    ):
        raise InvalidCell("persisted application HTTP route graph drifted")
    active_projected_routes = {
        key: root
        for key, root in projected_routes.items()
        if key not in _RETIRED_APPLICATION_HTTP_ROUTE_KEYS
    }
    ensure_cloud_session_protocol(
        store, prefix="app:cloud-session-protocol"
    )
    snapshot = store.snapshot()
    cloud_session_protocol = project_cloud_session_protocol(
        snapshot, prefix="app:cloud-session-protocol"
    )
    browser_session_prefix = "app:browser-session-protocol"
    browser_session_root = browser_session_prefix + ":root"
    if browser_session_root in snapshot.cells:
        browser_session_protocol = project_browser_session_protocol(
            snapshot, prefix=browser_session_prefix
        )
    else:
        browser_session_protocol = bootstrap_browser_session_protocol(
            store, prefix=browser_session_prefix
        )
        snapshot = store.snapshot()
    application_browser_members = [
        member.participant_id for member in read_relation(
            snapshot, application_root, budget=100_000
        )
        if member.role_id == roles["member"]
        and member.participant_id == browser_session_protocol.root_id
    ]
    if not application_browser_members:
        browser_protocol_patch = prepare_append_relation_members(
            snapshot,
            application_root,
            ((roles["member"], browser_session_protocol.root_id),),
            budget=100_000,
        )
        store.commit(
            snapshot.revision,
            create=browser_protocol_patch.create,
            replace=browser_protocol_patch.replace,
        )
        snapshot = store.snapshot()
    elif len(application_browser_members) != 1:
        raise InvalidCell(
            "persisted browser-session protocol membership drifted"
        )
    ownership_prefix = "app:exclusive-ownership-protocol"
    ownership_root = ownership_prefix + ":root"
    if ownership_root in snapshot.cells:
        ownership_protocol = project_ownership_protocol(
            snapshot, prefix=ownership_prefix
        )
    else:
        ownership_protocol = bootstrap_ownership_protocol(
            store, prefix=ownership_prefix
        )
        snapshot = store.snapshot()
    application_ownership_members = [
        member.participant_id for member in read_relation(
            snapshot, application_root, budget=100_000
        )
        if member.role_id == roles["member"]
        and member.participant_id == ownership_protocol.root_id
    ]
    if not application_ownership_members:
        ownership_protocol_patch = prepare_append_relation_members(
            snapshot,
            application_root,
            ((roles["member"], ownership_protocol.root_id),),
            budget=100_000,
        )
        store.commit(
            snapshot.revision,
            create=ownership_protocol_patch.create,
            replace=ownership_protocol_patch.replace,
        )
        snapshot = store.snapshot()
    elif len(application_ownership_members) != 1:
        raise InvalidCell(
            "persisted ownership protocol membership drifted"
        )
    value_graph_prefix = "app:value-graph-protocol"
    value_graph_root = value_graph_prefix + ":root"
    if value_graph_root in snapshot.cells:
        value_graph_protocol = project_value_graph_protocol(
            snapshot, prefix=value_graph_prefix
        )
    else:
        value_graph_protocol = bootstrap_value_graph_protocol(
            store, prefix=value_graph_prefix
        )
        snapshot = store.snapshot()
    application_value_graph_members = [
        member.participant_id for member in read_relation(
            snapshot, application_root, budget=100_000
        )
        if member.role_id == roles["member"]
        and member.participant_id == value_graph_protocol.root_id
    ]
    if not application_value_graph_members:
        value_graph_protocol_patch = prepare_append_relation_members(
            snapshot,
            application_root,
            ((roles["member"], value_graph_protocol.root_id),),
            budget=100_000,
        )
        store.commit(
            snapshot.revision,
            create=value_graph_protocol_patch.create,
            replace=value_graph_protocol_patch.replace,
        )
        snapshot = store.snapshot()
    elif len(application_value_graph_members) != 1:
        raise InvalidCell(
            "persisted value-graph protocol membership drifted"
        )
    device_custody_prefix = "app:device-custody-protocol"
    device_custody_root = device_custody_prefix + ":root"
    if device_custody_root in snapshot.cells:
        device_custody_protocol = project_device_custody_protocol(
            snapshot, prefix=device_custody_prefix
        )
    else:
        device_custody_protocol = bootstrap_device_custody_protocol(
            store, prefix=device_custody_prefix
        )
        snapshot = store.snapshot()
    application_device_members = [
        member.participant_id for member in read_relation(
            snapshot, application_root, budget=100_000
        )
        if member.role_id == roles["member"]
        and member.participant_id == device_custody_protocol.root_id
    ]
    if not application_device_members:
        device_protocol_patch = prepare_append_relation_members(
            snapshot,
            application_root,
            ((roles["member"], device_custody_protocol.root_id),),
            budget=100_000,
        )
        store.commit(
            snapshot.revision,
            create=device_protocol_patch.create,
            replace=device_protocol_patch.replace,
        )
        snapshot = store.snapshot()
    elif len(application_device_members) != 1:
        raise InvalidCell(
            "persisted device-custody protocol membership drifted"
        )
    runtime_presence_prefix = "app:runtime-presence-protocol"
    runtime_presence_root = runtime_presence_prefix + ":root"
    if runtime_presence_root in snapshot.cells:
        runtime_presence_protocol = project_runtime_presence_protocol(
            snapshot, prefix=runtime_presence_prefix, store=store
        )
    else:
        runtime_presence_protocol = bootstrap_runtime_presence_protocol(
            store, prefix=runtime_presence_prefix
        )
        snapshot = store.snapshot()
    application_runtime_presence_members = [
        member.participant_id for member in read_relation(
            snapshot, application_root, budget=100_000
        )
        if member.role_id == roles["member"]
        and member.participant_id == runtime_presence_protocol.root_id
    ]
    if not application_runtime_presence_members:
        runtime_presence_patch = prepare_append_relation_members(
            snapshot,
            application_root,
            ((roles["member"], runtime_presence_protocol.root_id),),
            budget=100_000,
        )
        store.commit(
            snapshot.revision,
            create=runtime_presence_patch.create,
            replace=runtime_presence_patch.replace,
        )
        snapshot = store.snapshot()
    elif len(application_runtime_presence_members) != 1:
        raise InvalidCell(
            "persisted runtime-presence protocol membership drifted"
        )
    baboom_activity_prefix = "app:baboom-activity-protocol"
    baboom_activity_root = baboom_activity_prefix + ":root"
    if baboom_activity_root in snapshot.cells:
        baboom_activity_protocol = project_baboom_activity_protocol(
            snapshot,
            prefix=baboom_activity_prefix,
            activity_storage=runtime_presence_protocol.lease_storage,
            store=store,
        )
    else:
        baboom_activity_protocol = bootstrap_baboom_activity_protocol(
            store,
            prefix=baboom_activity_prefix,
            activity_storage=runtime_presence_protocol.lease_storage,
        )
        snapshot = store.snapshot()
    application_baboom_activity_members = [
        member.participant_id for member in read_relation(
            snapshot, application_root, budget=100_000
        )
        if member.role_id == roles["member"]
        and member.participant_id == baboom_activity_protocol.root_id
    ]
    if not application_baboom_activity_members:
        baboom_activity_patch = prepare_append_relation_members(
            snapshot,
            application_root,
            ((roles["member"], baboom_activity_protocol.root_id),),
            budget=100_000,
        )
        store.commit(
            snapshot.revision,
            create=baboom_activity_patch.create,
            replace=baboom_activity_patch.replace,
        )
        snapshot = store.snapshot()
    elif len(application_baboom_activity_members) != 1:
        raise InvalidCell(
            "persisted BABOOM activity protocol membership drifted"
        )
    baboom_meeting_notes_prefix = "app:baboom-meeting-notes-protocol"
    baboom_meeting_notes_root = baboom_meeting_notes_prefix + ":root"
    if baboom_meeting_notes_root in snapshot.cells:
        baboom_meeting_notes_protocol = project_baboom_meeting_notes_protocol(
            snapshot, prefix=baboom_meeting_notes_prefix
        )
    else:
        baboom_meeting_notes_protocol = bootstrap_baboom_meeting_notes_protocol(
            store, prefix=baboom_meeting_notes_prefix
        )
        snapshot = store.snapshot()
    application_baboom_meeting_notes_members = [
        member.participant_id for member in read_relation(
            snapshot, application_root, budget=100_000
        )
        if member.role_id == roles["member"]
        and member.participant_id == baboom_meeting_notes_protocol.root_id
    ]
    if not application_baboom_meeting_notes_members:
        baboom_meeting_notes_patch = prepare_append_relation_members(
            snapshot,
            application_root,
            ((roles["member"], baboom_meeting_notes_protocol.root_id),),
            budget=100_000,
        )
        store.commit(
            snapshot.revision,
            create=baboom_meeting_notes_patch.create,
            replace=baboom_meeting_notes_patch.replace,
        )
        snapshot = store.snapshot()
    elif len(application_baboom_meeting_notes_members) != 1:
        raise InvalidCell(
            "persisted BABOOM meeting-notes protocol membership drifted"
        )
    baboom_meeting_note_publication_prefix = (
        "app:baboom-meeting-note-publication-protocol"
    )
    baboom_meeting_note_publication_root = (
        baboom_meeting_note_publication_prefix + ":root"
    )
    if baboom_meeting_note_publication_root in snapshot.cells:
        baboom_meeting_note_publication_protocol = (
            project_baboom_meeting_note_publication_protocol(
                snapshot, prefix=baboom_meeting_note_publication_prefix
            )
        )
    else:
        baboom_meeting_note_publication_protocol = (
            bootstrap_baboom_meeting_note_publication_protocol(
                store, prefix=baboom_meeting_note_publication_prefix
            )
        )
        snapshot = store.snapshot()
    application_baboom_meeting_note_publication_members = [
        member.participant_id for member in read_relation(
            snapshot, application_root, budget=100_000
        )
        if member.role_id == roles["member"]
        and member.participant_id == baboom_meeting_note_publication_protocol.root_id
    ]
    if not application_baboom_meeting_note_publication_members:
        baboom_meeting_note_publication_patch = prepare_append_relation_members(
            snapshot,
            application_root,
            ((roles["member"], baboom_meeting_note_publication_protocol.root_id),),
            budget=100_000,
        )
        store.commit(
            snapshot.revision,
            create=baboom_meeting_note_publication_patch.create,
            replace=baboom_meeting_note_publication_patch.replace,
        )
        snapshot = store.snapshot()
    elif len(application_baboom_meeting_note_publication_members) != 1:
        raise InvalidCell(
            "persisted BABOOM meeting-note publication protocol membership drifted"
        )
    compliance_root = _RUNTIME_COMPLIANCE_PROTOCOL_PREFIX + ":root"
    if compliance_root in snapshot.cells:
        compliance_protocol = project_compliance_protocol(
            snapshot, prefix=_RUNTIME_COMPLIANCE_PROTOCOL_PREFIX
        )
    else:
        compliance_protocol = bootstrap_compliance_protocol(
            store, prefix=_RUNTIME_COMPLIANCE_PROTOCOL_PREFIX
        )
        snapshot = store.snapshot()
    application_compliance_members = [
        member.participant_id for member in read_relation(
            snapshot, application_root, budget=100_000
        )
        if member.role_id == roles["member"]
        and member.participant_id in {
            compliance_protocol.root_id,
            compliance_protocol.registry_root,
        }
    ]
    if len(application_compliance_members) != len(
        set(application_compliance_members)
    ):
        raise InvalidCell("persisted compliance membership is duplicated")
    if set(application_compliance_members) != {
        compliance_protocol.root_id,
        compliance_protocol.registry_root,
    }:
        missing_compliance_roots = tuple(
            root for root in (
                compliance_protocol.root_id,
                compliance_protocol.registry_root,
            )
            if root not in application_compliance_members
        )
        compliance_patch = prepare_append_relation_members(
            snapshot,
            application_root,
            (
                (roles["member"], root)
                for root in missing_compliance_roots
            ),
            budget=100_000,
        )
        store.commit(
            snapshot.revision,
            create=compliance_patch.create,
            replace=compliance_patch.replace,
        )
        snapshot = store.snapshot()
    work_handoff_prefix = "app:work-handoff-protocol"
    work_handoff_root = work_handoff_prefix + ":root"
    if work_handoff_root in snapshot.cells:
        work_handoff_protocol = project_work_handoff_protocol(
            snapshot, prefix=work_handoff_prefix
        )
    else:
        work_handoff_protocol = bootstrap_work_handoff_protocol(
            store, prefix=work_handoff_prefix
        )
        snapshot = store.snapshot()
    application_work_handoff_members = [
        member.participant_id for member in read_relation(
            snapshot, application_root, budget=100_000
        )
        if member.role_id == roles["member"]
        and member.participant_id == work_handoff_protocol.root_id
    ]
    if not application_work_handoff_members:
        work_handoff_patch = prepare_append_relation_members(
            snapshot,
            application_root,
            ((roles["member"], work_handoff_protocol.root_id),),
            budget=100_000,
        )
        store.commit(
            snapshot.revision,
            create=work_handoff_patch.create,
            replace=work_handoff_patch.replace,
        )
        snapshot = store.snapshot()
    elif len(application_work_handoff_members) != 1:
        raise InvalidCell(
            "persisted work-handoff protocol membership drifted"
        )
    work_claim_transfer_prefix = "app:work-claim-transfer-protocol"
    work_claim_transfer_root = work_claim_transfer_prefix + ":root"
    if work_claim_transfer_root in snapshot.cells:
        work_claim_transfer_protocol = project_work_claim_transfer_protocol(
            snapshot, prefix=work_claim_transfer_prefix
        )
    else:
        work_claim_transfer_protocol = bootstrap_work_claim_transfer_protocol(
            store, prefix=work_claim_transfer_prefix
        )
        snapshot = store.snapshot()
    application_work_claim_transfer_members = [
        member.participant_id for member in read_relation(
            snapshot, application_root, budget=100_000
        )
        if member.role_id == roles["member"]
        and member.participant_id == work_claim_transfer_protocol.root_id
    ]
    if not application_work_claim_transfer_members:
        work_claim_transfer_patch = prepare_append_relation_members(
            snapshot,
            application_root,
            ((roles["member"], work_claim_transfer_protocol.root_id),),
            budget=100_000,
        )
        store.commit(
            snapshot.revision,
            create=work_claim_transfer_patch.create,
            replace=work_claim_transfer_patch.replace,
        )
        snapshot = store.snapshot()
    elif len(application_work_claim_transfer_members) != 1:
        raise InvalidCell(
            "persisted work-claim-transfer protocol membership drifted"
        )
    cde_signing_prefix = "app:cde-signing-authority-protocol"
    cde_signing_root = cde_signing_prefix + ":root"
